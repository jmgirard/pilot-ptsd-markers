redcap_dir <- Sys.getenv("CMBA_REDCAP_DIR", r"(Z:\CMBA_PTSD\1_RAWDATA\redCAP)")

redcap_vars <- c(
  "record_id", "race", "ethnicity", "sex", "age", "income", "education",
  "phq_total", "pcl5_m_total", "caps_mnth_total", "caps_mnth_ptsd",
  "suds_account", "suds_impact"
)

read_redcap <- function(file, group) {
  read_csv(
    file.path(redcap_dir, file),
    col_types = cols(.default = col_character())
  ) |>
    select(all_of(redcap_vars)) |>
    summarize(
      .by = record_id,
      group = group,
      across(everything(), \(x) first(x, na_rm = TRUE))
    )
}

demo_all <-

fmt_p <- function(p) {
  if_else(p < .001, "p < .001", paste("p =", sub("^0", "", sprintf("%.3f", p))))
}

tabulate_var <- function(data, var) {
  tab <- table(droplevels(data[[var]]), data$PTSD)
  pct <- 100 * prop.table(tab, 2)
  tibble(
    variable = var,
    level = rownames(tab),
    n_ptsd = tab[, "PTSD"],
    pct_ptsd = pct[, "PTSD"],
    n_ctrl = tab[, "Control"],
    pct_ctrl = pct[, "Control"]
  )
}

# Sex: chi-square test
sex_test <- chisq.test(table(droplevels(demo_all$Sex), demo_all$PTSD), correct = FALSE)

# Race and ethnicity: Fisher's exact test because some expected counts are < 5
race_test <- fisher.test(table(droplevels(demo_all$Race), demo_all$PTSD))
ethnicity_test <- fisher.test(table(droplevels(demo_all$Ethnicity), demo_all$PTSD))

# Income: Fisher's exact test, excluding "Prefer not to say"
income_data <- filter(demo_all, Income != "Prefer not to say")
income_test <- fisher.test(table(droplevels(income_data$Income), income_data$PTSD))

categorical_tests <- tibble(
  variable = c("Sex", "Race", "Ethnicity", "Income"),
  test = c(
    sprintf("χ²(%d) = %.2f, %s", sex_test$parameter, sex_test$statistic, fmt_p(sex_test$p.value)),
    paste("Fisher's exact,", fmt_p(race_test$p.value)),
    paste("Fisher's exact,", fmt_p(ethnicity_test$p.value)),
    paste("Fisher's exact,", fmt_p(income_test$p.value))
  )
)

categorical_table <-
  map(categorical_tests$variable, \(v) tabulate_var(demo_all, v)) |>
  list_rbind()

continuous_vars <- c(
  "Age" = "Age (years)",
  "Education" = "Education (years)",
  "SUDS_Account" = "SUDS, Account",
  "SUDS_Impact" = "SUDS, Impact",
  "PCL5" = "PCL-5",
  "CAPS5" = "CAPS-5"
)

# Mann-Whitney U tests with rank-biserial correlations (positive = higher in PTSD group)
continuous_table <-
  imap(continuous_vars, \(label, var) {
    x <- na.omit(demo_all[[var]][demo_all$PTSD == "PTSD"])
    y <- na.omit(demo_all[[var]][demo_all$PTSD == "Control"])
    w <- wilcox.test(x, y, exact = FALSE)
    tibble(
      variable = label,
      m_ptsd = mean(x), sd_ptsd = sd(x),
      m_ctrl = mean(y), sd_ctrl = sd(y),
      U = unname(w$statistic),
      p = w$p.value,
      r = unname(2 * w$statistic / (length(x) * length(y)) - 1)
    )
  }) |>
  list_rbind()

continuous_table |>
  gt() |>

  fmt_number(columns = U, decimals = 1) |>

  cols_label(
    variable = "", m_ptsd = "M (PTSD)", sd_ptsd = "SD (PTSD)",
    m_ctrl = "M (Control)", sd_ctrl = "SD (Control)"
  )

table_1 <- bind_rows(
  tibble(row = "Categorical variables", c1 = "Count", c2 = "Percent", c3 = "Count", c4 = "Percent", test = "", type = "header"),
  map(categorical_tests$variable, \(v) {
    bind_rows(
      tibble(row = v, test = categorical_tests$test[categorical_tests$variable == v], type = "variable"),
      categorical_table |>
        filter(variable == v) |>
        transmute(
          row = level,
          c1 = as.character(n_ptsd),
          c2 = sprintf("%.1f%%", pct_ptsd),
          c3 = as.character(n_ctrl),
          c4 = sprintf("%.1f%%", pct_ctrl),
          test = "",
          type = "level"
        )
    )
  }) |> list_rbind(),
  tibble(row = "Continuous variables", c1 = "M", c2 = "SD", c3 = "M", c4 = "SD", test = "r", type = "header"),
  continuous_table |>
    transmute(
      row = variable,
      c1 = sprintf("%.2f", m_ptsd),
      c2 = sprintf("%.2f", sd_ptsd),
      c3 = sprintf("%.2f", m_ctrl),
      c4 = sprintf("%.2f", sd_ctrl),
      test = sub("-", "−", sprintf("%.2f", r)),
      type = "level"
    )
) |>
  mutate(across(c1:c4, \(x) replace_na(x, "")))
