read_redcap <- function(file, group) {
  read_csv(
    file.path(redcap_dir, file),
    col_types = cols(.default = col_character())
  ) |>
    select(all_of(redcap_vars)) |>
    summarize(
      .by = record_id,
      group = group,
      across(everything(), \(x) first(x, na_rm = TRUE))
    )
}

# Formatting helpers (APA style: no leading zero for p and bounded effect sizes)
fmt_p <- function(p) {
  if_else(p < .001, "< .001", sub("^0", "", sprintf("%.3f", p)))
}
fmt_es <- function(x) {
  sub("^(-?)0", "\\1", sprintf("%.2f", x)) |> sub(pattern = "^-", replacement = "−")
}

# Mann-Whitney U test with rank-biserial correlation (positive = higher in PTSD group)
mwu_test <- function(data, var) {
  data <- filter(data, !is.na(.data[[var]]))
  x <- data[[var]][data$PTSD == "PTSD"]
  y <- data[[var]][data$PTSD == "Control"]
  w <- wilcox.test(x, y, exact = FALSE)
  tibble(
    test = "Mann-Whitney U",
    statistic = sprintf("U = %.1f", w$statistic),
    p = w$p.value,
    es_name = "r",
    es = unname(2 * w$statistic / (length(x) * length(y)) - 1),
    n = length(x) + length(y)
  )
}

# Chi-square or Fisher's exact test with Cramér's V
nominal_test <- function(data, var, exact) {
  tab <- table(droplevels(data[[var]]), data$PTSD)
  chi <- suppressWarnings(chisq.test(tab, correct = FALSE))
  v <- sqrt(unname(chi$statistic) / (sum(tab) * (min(dim(tab)) - 1)))
  result <- if (exact) {
    tibble(test = "Fisher's exact", statistic = "", p = fisher.test(tab)$p.value)
  } else {
    tibble(
      test = "Chi-square",
      statistic = sprintf("χ²(%d) = %.2f", chi$parameter, chi$statistic),
      p = chi$p.value
    )
  }
  mutate(result, es_name = "V", es = v, n = sum(tab), min_expected = min(chi$expected))
}

comparisons <- bind_rows(
  # Expected counts are all >= 5 for sex
  nominal_test(demo_all, "Sex", exact = FALSE) |> mutate(variable = "Sex"),
  # Some expected counts are < 5 for race and ethnicity
  nominal_test(demo_all, "Race", exact = TRUE) |> mutate(variable = "Race"),
  nominal_test(filter(demo_all, Ethnicity != "Unknown"), "Ethnicity", exact = TRUE) |>
    mutate(variable = "Ethnicity"),
  # Income is ordinal, so compare the ordered income brackets
  demo_all |>
    filter(Income != "Prefer not to say") |>
    mutate(Income_rank = as.integer(droplevels(Income))) |>
    mwu_test("Income_rank") |>
    mutate(variable = "Income"),
  mwu_test(demo_all, "Age") |> mutate(variable = "Age (years)"),
  mwu_test(demo_all, "Education") |> mutate(variable = "Education (years)"),
  mwu_test(demo_all, "SUDS_Account") |> mutate(variable = "SUDS, Account"),
  mwu_test(demo_all, "SUDS_Impact") |> mutate(variable = "SUDS, Impact"),
  mwu_test(demo_all, "PCL5") |> mutate(variable = "PCL-5")
  # CAPS-5 is not tested because the groups were defined by the CAPS-5 diagnosis
) |>
  relocate(variable)

tabulate_var <- function(var) {
  tab <- table(droplevels(demo_all[[var]]), demo_all$PTSD)
  pct <- 100 * prop.table(tab, 2)
  tibble(
    row = rownames(tab),
    c1 = as.character(tab[, "PTSD"]),
    c2 = sprintf("%.1f%%", pct[, "PTSD"]),
    c3 = as.character(tab[, "Control"]),
    c4 = sprintf("%.1f%%", pct[, "Control"]),
    type = "level"
  )
}

describe_var <- function(var, label) {
  x <- na.omit(demo_all[[var]][demo_all$PTSD == "PTSD"])
  y <- na.omit(demo_all[[var]][demo_all$PTSD == "Control"])
  tibble(
    row = label,
    c1 = sprintf("%.2f", mean(x)), c2 = sprintf("%.2f", sd(x)),
    c3 = sprintf("%.2f", mean(y)), c4 = sprintf("%.2f", sd(y)),
    type = "continuous"
  )
}
