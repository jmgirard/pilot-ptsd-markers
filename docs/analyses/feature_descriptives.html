read_features <- function(modality) {
  read.csv(file.path("..", "data", paste0(modality, ".csv"))) |>
    as_tibble()
}

id_cols <- c("participant", "source", "ptsd", "demo_age", "demo_sex", "demo_nonwhite", "task")

df_features <-
  read_features("verbal") |>
  full_join(read_features("vocal"), by = id_cols) |>
  full_join(read_features("visual"), by = id_cols) |>
  mutate(
    group = factor(ptsd, levels = c(1, 0), labels = c("PTSD", "Control")),
    task = factor(task, levels = c("Account", "Impact"))
  )

cell_stats <-
  df_features |>
  pivot_longer(all_of(spec$var), names_to = "var") |>
  summarize(
    .by = c(var, group, task),
    m = mean(value, na.rm = TRUE),
    s = sd(value, na.rm = TRUE)
  ) |>
  left_join(spec, by = "var") |>
  mutate(cell = sprintf("%.*f (%.*f)", digits, m, digits, s))

table_s1 <-
  cell_stats |>
  select(modality, var, label, group, task, cell) |>
  pivot_wider(names_from = c(group, task), values_from = cell, names_sep = "_") |>
  arrange(match(var, spec$var)) |>
  select(modality, label, PTSD_Account, Control_Account, PTSD_Impact, Control_Impact)

n_group <- df_features |> distinct(participant, group) |> count(group) |> deframe()

table_s1 |>
  gt(groupname_col = "modality", rowname_col = "label") |>
  tab_header(
    title = "Table S1",
    subtitle = "Means and Standard Deviations for Behavioral Features by Group and Task"
  ) |>
  tab_spanner(label = "Trauma Account", columns = c(PTSD_Account, Control_Account)) |>
  tab_spanner(label = "Impact Statement", columns = c(PTSD_Impact, Control_Impact)) |>
  cols_label(
    PTSD_Account = sprintf("PTSD (n = %d)", n_group[["PTSD"]]),
    Control_Account = sprintf("Control (n = %d)", n_group[["Control"]]),
    PTSD_Impact = sprintf("PTSD (n = %d)", n_group[["PTSD"]]),
    Control_Impact = sprintf("Control (n = %d)", n_group[["Control"]])
  ) |>
  cols_align(align = "center", columns = -label) |>
  tab_source_note(md(paste(
    "*Note.* Values are *M* (*SD*) in each feature's original units.",
    "LIWC-22 categories are percentages of total words.",
    "Sentiment is the LLM-predicted rating of each transcript (1 = very negative, 7 = very positive).",
    "Action unit (AU) values are OpenFace intensity estimates (0–5) averaged over frames.",
    "Head pose values are mean rotations and head motion values are the root mean square of",
    "successive frame-to-frame differences in rotation, in radians.",
    "The models analyze LIWC percentages divided by 100 and AU intensities divided by 5."
  )))
